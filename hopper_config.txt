# BLACK ICE v2 — Channel Hopper Configuration
#
# Periodically switch the Wi-Fi adapter's listening channel so the
# passive scanner can observe APs across the band, not just one
# parked channel. Off by default. You can also enable this at launch
# with --hop or --channel-hop (PowerShell):
#   java -jar target\jwscan-2.0.0-jar-with-dependencies.jar 8080 --hop
#
# Requires Administrator (Windows) or root (Linux) and a
# monitor-mode-capable adapter. Windows uses netsh; Linux uses iw.

# 0 = off (default), 1 = on
enabled=0

# Milliseconds to dwell on each channel.
# 150ms catches at least one beacon per AP (APs beacon every 100ms).
dwell_ms=150

# Include 5 GHz channels (1=yes, 0=no).
band_5ghz=1

# Include 6 GHz / Wi-Fi 6E channels (1=yes, 0=no).
band_6ghz=0

# Hopping strategy:
#   SEQUENTIAL  — visit 1,2,3,...13,36,40,... in order
#   INTERLEAVED — non-overlapping 2.4GHz first (1,6,11), then rest
#   SMART       — weighted: more time on channels with known APs (recommended)
strategy=SMART
